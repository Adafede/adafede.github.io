% Tagging gives the PDF a structure tree (headings, lists, paragraphs) that a
% screen reader can navigate rather than a wall of text. `lang` is the document
% language, which is what /Lang in the file descriptor is set from, and which
% decides hyphenation. Both are PDF/UA requirements; without them the file is
% technically valid and practically unreadable to assistive technology.
%
% DocumentMetadata has to be the first thing after \documentclass. pdfTeX of
% TeX Live 2020 or newer is required for tagging=on.

% Language declaration. It is what hyphenation follows and, through babel,
% what hyperref writes into /Lang — the document language a screen reader
% announces. Tagged PDF is not offered by this TeX, which predates
% \DocumentMetadata, so no structure tree is available here; /Lang and the
% title metadata are what this install can provide.
\usepackage[english]{babel}
% /Lang (document language a screen reader announces, and what babel uses for
% hyphenation) is set in the hypersetup block below via `pdflang`. Declaring it
% once avoids a "Multiple definitions ... for key /Lang" PDF warning.

% create a coloured box with arrow and title as cv section headline
% param 1: section title
%
% Visual only: a centred boxed title. The PDF outline/bookmark tree is built by
% the `Header` filter in cv/styles.lua, which knows each heading's level and so
% can nest entry titles under their section; \cvsection itself stays level-
% agnostic.
\newcommand{\cvsection}[1]
{
	\begin{center}
		\large\textcolor{sectcol}{\textbf{#1}}
	\end{center}
}

%
\newcommand{\metasection}[2]
{
\footnotesize{#2} \hspace*{\fill} \footnotesize{#1}\\[1pt]
}

%----------------------------------------------------------------------------------------
%	 CV EVENT
%----------------------------------------------------------------------------------------

% creates a stretched box as cv entry headline followed by some paragraphs about
% the work you did
% param 1:	event time i.e. 2014 or 2011-2014 etc.
% param 2:	event name (what did you do?)
% param 3:	institution (where did you work / study)
% param 4:	list of paragraphs
%
\newcommand{\cvevent}[4]
{

\begin{tabular*}{1\textwidth}{p{13.6cm}  x{3.9cm}}
	\textbf{#2} - \textcolor{bgcol}{#3} &   \vspace{2.5pt}\textcolor{sectcol}{#1}
\end{tabular*}

\vspace{-8pt}
\textcolor{softcol}{\hrule}
\vspace{6pt}
\foreach \desc in {#4}{
	*  \desc\\[3pt]
}

\vspace{3pt}
}

% creates a stretched box as
\newcommand{\cveventmeta}[2]
{
	\mbox{\mystrut \hspace{87pt}\textit{#1}}\\
	#2
}
